query($owner:String!,$repo:String!,$pr:Int!) {
  repository(owner:$owner,name:$repo) {
    pullRequest(number:$pr) {
      reviewThreads(last:30) {
        nodes {
          id
          isResolved
          isOutdated
          comments(first:1) {
            nodes {
              body
              author { login }
              path
              line
            }
          }
        }
      }
    }
  }
}
